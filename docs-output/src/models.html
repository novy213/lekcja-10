Data models for the apartment management system.

 57class Apartment(BaseModel):
 58    """Apartment definition with identity, location, and room structure."""
 59
 60    key: str = Field(description="Unique apartment key used across datasets.")
 61    name: str = Field(description="Display name of the apartment.")
 62    location: str = Field(description="Address or location description.")
 63    area_m2: float = Field(description="Total apartment area in square meters.")
 64    rooms: dict[str, Room] = Field(
 65        description="Dictionary of rooms keyed by room identifier.",
 66    )
 67
 68    @staticmethod
 69    def from_json_file(file_path: str) -> dict[str, "Apartment"]:
 70        """Load apartments from a JSON file.
 71
 72        Args:
 73        ----
 74            file_path (str): Path to the JSON file containing apartment data.
 75
 76        Returns:
 77        -------
 78            dict[str, Apartment]: Apartments indexed by apartment key.
 79
 80        Example:
 81        -------
 82            >>> apartments = Apartment.from_json_file("data/apartments.json")
 83            >>> "A1" in apartments
 84            True
 85
 86        """
 87        data = None
 88        with open(file_path, encoding="utf-8") as file:
 89            data = json.load(file)
 90        assert isinstance(data, dict), "Expected a dictionary of apartments"
 91        return {key: Apartment(**apartment) for key, apartment in data.items()}
