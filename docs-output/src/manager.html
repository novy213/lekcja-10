175    def get_apartment_costs(
176        self,
177        apartment_key: str,
178        year: int = None,
179        month: int = None,
180    ) -> float | None:
181        """Calculate total billed costs for an apartment.
182
183        Args:
184        ----
185            apartment_key (str): Apartment identifier.
186            year (int | None): Optional settlement year filter.
187            month (int | None): Optional settlement month filter.
188
189        Returns:
190        -------
191            float | None: Total bill amount in PLN, or ``None`` if the
192            apartment key is unknown.
193
194        Raises:
195        ------
196            ValueError: If ``month`` is outside the ``1..12`` range.
197
198        Example:
199        -------
200            >>> from src.models import Parameters
201            >>> manager = Manager(Parameters())
202            >>> manager.get_apartment_costs("A1", year=2023, month=1) >= 0
203            True
204
205        """
206        if month is not None and (month < 1 or month > 12):
207            raise ValueError("Month must be between 1 and 12")
208        if apartment_key not in self.apartments:
209            return None
210        total_cost = 0.0
211        for bill in self.bills:
212            if (
213                bill.apartment == apartment_key
214                and (year is None or bill.settlement_year == year)
215                and (month is None or bill.settlement_month == month)
216            ):
217                total_cost += bill.amount_pln
218        return total_cost

268    def create_tenants_settlements(
269        self,
270        apartment_settlement: ApartmentSettlement,
271    ) -> list[TenantSettlement] | None:
272        """Split an apartment settlement evenly across its tenants.
273
274        Args:
275        ----
276            apartment_settlement (ApartmentSettlement): Apartment summary used
277                as the source for tenant shares.
278
279        Returns:
280        -------
281            list[TenantSettlement] | None: Tenant-level settlements, an empty
282            list when no tenants are assigned, or ``None`` if the apartment is
283            unknown.
284
285        Raises:
286        ------
287            ValueError: If the settlement month is outside the ``1..12`` range.
288
289        Example:
290        -------
291            >>> from src.models import Parameters
292            >>> manager = Manager(Parameters())
293            >>> settlement = manager.get_settlement("A1", year=2023, month=1)
294            >>> output = manager.create_tenants_settlements(settlement)
295            >>> isinstance(output, list)
296            True
297
298        """
299        if apartment_settlement.month < 1 or apartment_settlement.month > 12:
300            raise ValueError("Month must be between 1 and 12")
301        if apartment_settlement.apartment not in self.apartments:
302            return None
303        tenants_in_apartment = [
304            tenant
305            for tenant in self.tenants.values()
306            if tenant.apartment == apartment_settlement.apartment
307        ]
308        if not tenants_in_apartment:
309            return []
310
311        return [
312            TenantSettlement(
313                tenant=tenant.name,
314                apartment_settlement=apartment_settlement.key,
315                month=apartment_settlement.month,
316                year=apartment_settlement.year,
317                total_due_pln=apartment_settlement.total_due_pln
318                / len(tenants_in_apartment),
319            )
320            for tenant in tenants_in_apartment
321        ]

323    def get_debtors(self, apartment_key: str, year: int, month: int) -> list[str]:
324        """Return tenant names whose payments do not cover the monthly share.
325
326        Args:
327        ----
328            apartment_key (str): Apartment identifier.
329            year (int): Settlement year.
330            month (int): Settlement month.
331
332        Returns:
333        -------
334            list[str]: Tenant names with insufficient payments for the period.
335
336        Raises:
337        ------
338            ValueError: If ``month`` is outside the ``1..12`` range.
339
340        Example:
341        -------
342            >>> from src.models import Parameters
343            >>> manager = Manager(Parameters())
344            >>> debtors = manager.get_debtors("A1", year=2023, month=1)
345            >>> isinstance(debtors, list)
346            True
347
348        """
349        if month < 1 or month > 12:
350            raise ValueError("Month must be between 1 and 12")
351        output = []
352        settlement = self.get_settlement(apartment_key, year, month)
353        tenant_settlements = self.create_tenants_settlements(settlement)
354
355        for tenant_settlement in tenant_settlements:
356            tenant_transfers = [
357                transfer
358                for transfer in self.transfers
359                if self.tenants[transfer.tenant].name == tenant_settlement.tenant
360                and transfer.settlement_year == year
361                and transfer.settlement_month == month
362            ]
363            total_paid = sum(
364                transfer.amount_pln
365                for transfer in tenant_transfers
366                if transfer.settlement_year == year
367                and transfer.settlement_month == month
368            )
369            if total_paid < tenant_settlement.total_due_pln:
370                output.append(tenant_settlement.tenant)
371        return output

401    def check_deposits(self) -> float:
402        """Compare collected tenant deposits with required deposit values.
403
404        Returns
405        -------
406            float: Difference between paid deposits and expected deposits.
407            Positive values mean overpayment, negative values mean shortage.
408
409        Example:
410        -------
411            >>> from src.models import Parameters
412            >>> manager = Manager(Parameters())
413            >>> isinstance(manager.check_deposits(), float)
414            True
415
416        """
417        total_deposits = 0.0
418        total_due = 0.0
419        for _, tenant in self.tenants.items():
420            total_deposits += sum(
421                transfer.amount_pln
422                for transfer in self.transfers
423                if self.tenants[transfer.tenant].name == tenant.name
424                and transfer.type == "deposit"
425            )
426            total_due += tenant.deposit_pln
427
428        return total_deposits - total_due

537    def check_transfers_tenant(self) -> bool:
538        """Validate transfer ownership and agreement date consistency.
539
540        Returns
541        -------
542            bool: ``True`` when every transfer references an existing tenant and
543            falls within the tenant agreement period.
544
545        """
546        for transfer in self.transfers:
547            if transfer.tenant not in self.tenants:
548                return False
549            if (
550                transfer.settlement_year is not None
551                and transfer.settlement_month is not None
552            ):
553                agreement_from = self.tenants[transfer.tenant].date_agreement_from
554                agreement_from = datetime.strptime(agreement_from, "%Y-%m-%d").date()
555                agreement_to = self.tenants[transfer.tenant].date_agreement_to
556                agreement_to = datetime.strptime(agreement_to, "%Y-%m-%d").date()
557                if (transfer.settlement_year < agreement_from.year) or (
558                    transfer.settlement_year > agreement_to.year
559                ):
560                    return False
561
562        return True
